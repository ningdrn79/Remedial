Konfigurasi Google Sheets (Sinkronisasi Serentak)

Fitur ini memungkinkan puluhan/ratusan siswa mengerjakan remedial bersamaan di perangkat laptop/HP masing-masing, dan hasilnya otomatis terpusat masuk ke satu file Google Sheets Anda!

1 Buka Google Spreadsheet & Apps Script

Buka Google Sheets baru > Menu Extensions (Ekstensi) > Apps Script.

2 Salin & Tempel Kode Script Ini

// Google Apps Script untuk Remedial Serentak SDS Cinta Kasih Tzu Chi
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "ID Siswa", "Nama Lengkap", "Kelas", "No Absen", "Nilai Awal", "Nilai Akhir", "Predikat", "Status", "Durasi", "Jawaban PG", "Jawaban Essay"]);
    }
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.studentId || "",
      data.studentName || "",
      data.grade || "",
      data.absen || "-",
      data.nilaiAwal !== undefined ? data.nilaiAwal : (data.initialScore || 0),
      data.score || 0,
      data.predicate || "",
      data.status || "",
      data.duration || "-",
      JSON.stringify(data.pgAnswers || {}),
      JSON.stringify(data.essayAnswers || {})
    ]);
    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

    var idxDur = headers.indexOf("durasi");
    var idxPg = headers.indexOf("jawaban pg");

    function safeParse(val) {
      if (!val) return [];
      try {
        return typeof val === "string" && (val.charAt(0) === "{" || val.charAt(0) === "[") ? JSON.parse(val) : [];
      } catch (errParse) {
        return [];
      }
    }

    var records = [];
    for (var i = 1; i < rows.length; i++) {
      var r = rows[i];
      var isNewFormat = (idxAwal !== -1 || r.length >= 10);
      records.push({
        date: r[0],
        studentId: idxId !== -1 ? r[idxId] : r[1],
        studentName: idxName !== -1 ? r[idxName] : r[2],
        grade: idxGrade !== -1 ? r[idxGrade] : r[3],
        absen: idxAbsen !== -1 ? r[idxAbsen] : (isNewFormat ? r[4] : "-"),
        nilaiAwal: idxAwal !== -1 ? r[idxAwal] : (isNewFormat ? r[5] : 0),
        score: idxScore !== -1 ? r[idxScore] : (isNewFormat ? r[6] : r[4]),
        predicate: idxPred !== -1 ? r[idxPred] : (isNewFormat ? r[7] : r[5]),
        status: idxStatus !== -1 ? r[idxStatus] : (isNewFormat ? r[8] : r[6]),
        duration: idxDur !== -1 ? r[idxDur] : (isNewFormat ? r[9] : "-"),
        pgAnswers: safeParse(idxPg !== -1 ? r[idxPg] : r[7]),
        essayAnswers: safeParse(idxEssay !== -1 ? r[idxEssay] : r[8])
      });
    }
    return ContentService.createTextOutput(JSON.stringify(records))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
